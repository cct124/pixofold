cmake_minimum_required(VERSION 3.21)
project(pixofold_installer C RC)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows MSI actions require the MSVC toolchain")
endif()
function(pfx_target name)
  target_compile_definitions(${name} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_compile_options(${name} PRIVATE /W4 /WX /utf-8)
  target_link_libraries(${name} PRIVATE msi shell32 advapi32 ole32 uuid)
endfunction()
add_library(pixofold-installer SHARED installer.c cleanup.c installer.rc)
pfx_target(pixofold-installer)
target_link_options(pixofold-installer PRIVATE /MANIFEST:NO)
if(PFX_BUILD_TESTS)
  add_executable(cleanup-tests tests.c cleanup.c)
  pfx_target(cleanup-tests)
  add_executable(dialog-preview installer.c cleanup.c installer.rc)
  target_compile_definitions(dialog-preview PRIVATE PFX_DIALOG_PREVIEW=1)
  target_link_options(dialog-preview PRIVATE /MANIFEST:NO)
  pfx_target(dialog-preview)
  add_library(pixofold-installer-probe SHARED installer.c cleanup.c installer.rc)
  target_compile_definitions(pixofold-installer-probe PRIVATE PFX_INSTALLER_PROBE=1)
  target_link_options(pixofold-installer-probe PRIVATE /MANIFEST:NO)
  pfx_target(pixofold-installer-probe)
endif()
