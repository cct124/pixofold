cmake_minimum_required(VERSION 3.15)
project(pixofold_gifsicle_lab LANGUAGES C)

# Windows采用上游win32cfg配置和同一源文件列表，补齐1.96新增的kcolor.c。
# 不修改源码缓存；静态CRT与无线程/SIMD配置由此固定，其他平台用上游configure。
if(NOT GIFSICLE_SOURCE)
  message(FATAL_ERROR "Missing verified Gifsicle source")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
file(TO_CMAKE_PATH "${GIFSICLE_SOURCE}" GIFSICLE_SOURCE)
if(MSVC)
  configure_file("${GIFSICLE_SOURCE}/src/win32cfg.h" "${CMAKE_CURRENT_BINARY_DIR}/config.h" COPYONLY)
elseif(GIFSICLE_CONFIG)
  configure_file("${GIFSICLE_CONFIG}" "${CMAKE_CURRENT_BINARY_DIR}/config.h" COPYONLY)
else()
  message(FATAL_ERROR "Missing upstream configure output")
endif()
set(GIFSICLE_FILES clp fmalloc giffunc gifread gifunopt gifwrite kcolor merge optimize quantize support xform gifsicle)
list(TRANSFORM GIFSICLE_FILES PREPEND "${GIFSICLE_SOURCE}/src/")
list(TRANSFORM GIFSICLE_FILES APPEND ".c")
add_executable(gifsicle ${GIFSICLE_FILES})
target_compile_definitions(gifsicle PRIVATE HAVE_CONFIG_H _CONSOLE _CRT_SECURE_NO_WARNINGS)
target_include_directories(gifsicle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${GIFSICLE_SOURCE}/include")
add_library(bounded_gifsicle OBJECT ${GIFSICLE_FILES})
target_compile_definitions(bounded_gifsicle PRIVATE
  main=pixofold_gifsicle_main malloc=pixofold_gif_malloc realloc=pixofold_gif_realloc
  calloc=pixofold_gif_calloc free=pixofold_gif_free)
add_executable(pixofold-gif-helper ../../native/gif/entry.c ../../native/gif/budget.c
  $<TARGET_OBJECTS:bounded_gifsicle>)
foreach(target gifsicle bounded_gifsicle pixofold-gif-helper)
  set_property(TARGET ${target} PROPERTY C_STANDARD 11)
  target_compile_definitions(${target} PRIVATE HAVE_CONFIG_H _CONSOLE _CRT_SECURE_NO_WARNINGS)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${GIFSICLE_SOURCE}/include")
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /utf-8 /W2)
  else()
    target_compile_options(${target} PRIVATE -O2)
  endif()
endforeach()
if(NOT MSVC)
  target_link_libraries(gifsicle PRIVATE m)
  target_link_libraries(pixofold-gif-helper PRIVATE m)
endif()
